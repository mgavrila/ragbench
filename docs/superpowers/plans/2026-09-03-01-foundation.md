# RAGBench Foundation Implementation Plan (Plan 1 of 6)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the RAGBench monorepo foundation: workspace scaffold, Postgres+pgvector schema, job-queue worker, provider adapters (mock + Anthropic + OpenAI), auth with org tenancy, and a booting Next.js app with project CRUD.

**Architecture:** pnpm monorepo with `apps/web` (Next.js App Router, serves UI + API), `apps/worker` (pg-boss consumer), `packages/core` (pure domain logic: providers, registry), `packages/db` (Drizzle schema + client, shared by web and worker). One Postgres (pgvector image) is both datastore and queue. This plan ends with: `docker compose up postgres` + `pnpm dev` boots the stack, a user can sign up (org auto-created), log in, and create a project; the worker processes a test job; all tests green.

**Tech Stack:** TypeScript (strict), pnpm workspaces, Next.js (App Router) ≥15, Drizzle ORM + drizzle-kit, Postgres 17 (pgvector/pgvector image), pg-boss ≥10, Auth.js (next-auth v5), bcryptjs, Zod, Vitest, @anthropic-ai/sdk, openai.

**Spec:** `docs/superpowers/specs/2026-09-03-ragbench-design.md` (sections 2, 3, 9 are implemented by this plan; later plans cover ingestion, test-set generation, eval runs, attribution, and launch polish).

## Global Constraints

- TypeScript `strict: true` everywhere; no `any` except at validated JSON boundaries.
- Node ≥ 22. Package manager is pnpm; never npm/yarn.
- All timestamps `timestamptz`, all ids `uuid` default `gen_random_uuid()` (Drizzle `defaultRandom()`).
- Default LLM model id is exactly `claude-opus-5`; cheap option `claude-haiku-4-5`; embeddings `text-embedding-3-small` (1536 dims) and `text-embedding-3-large` (3072 dims). Never invent other model ids.
- Provider API keys come only from env (`ANTHROPIC_API_KEY`, `OPENAI_API_KEY`); never stored in DB.
- `packages/core` must not import from `packages/db`, `next`, or `pg-boss` — it stays pure (see spec §2).
- Tests never call real provider APIs; CI must pass with no keys set.
- Commit messages: conventional style (`feat:`, `chore:`, `test:`), no Co-Authored-By trailer.
- Test DB URL comes from `TEST_DATABASE_URL`, defaulting to `postgres://ragbench:ragbench@localhost:5432/ragbench`.

```ts
export const LLM_MODELS: Record<
  string,
  { provider: "anthropic"; inputPerMTok: number; outputPerMTok: number }
> = {
  "claude-opus-5": { provider: "anthropic", inputPerMTok: 5, outputPerMTok: 25 },
  "claude-haiku-4-5": { provider: "anthropic", inputPerMTok: 1, outputPerMTok: 5 },
};

export const EMBEDDING_MODELS: Record<
  string,
  { provider: "openai" | "mock"; dimension: number; pricePerMTok: number }
> = {
  "text-embedding-3-small": { provider: "openai", dimension: 1536, pricePerMTok: 0.02 },
  "text-embedding-3-large": { provider: "openai", dimension: 3072, pricePerMTok: 0.13 },
  "mock-embedding": { provider: "mock", dimension: 256, pricePerMTok: 0 },
};

export const DEFAULT_LLM = "claude-opus-5";
export const CHEAP_LLM = "claude-haiku-4-5";
export const DEFAULT_EMBEDDER = "text-embedding-3-small";

export function estimateLlmCostUsd(model: string, inputTokens: number, outputTokens: number): number {
  const m = LLM_MODELS[model];
  if (!m) throw new Error(`unknown LLM model: ${model}`);
  return (inputTokens * m.inputPerMTok + outputTokens * m.outputPerMTok) / 1_000_000;
}

export function estimateEmbeddingCostUsd(model: string, tokens: number): number {
  const m = EMBEDDING_MODELS[model];
  if (!m) throw new Error(`unknown embedding model: ${model}`);
  return (tokens * m.pricePerMTok) / 1_000_000;
}
```

```ts
import pg from "pg";
import { drizzle } from "drizzle-orm/node-postgres";
import { migrate } from "drizzle-orm/node-postgres/migrator";
import { fileURLToPath } from "url";
import path from "path";
import * as schema from "./schema";

const migrationsFolder = path.join(path.dirname(fileURLToPath(import.meta.url)), "..", "migrations");

export function createDb(url: string) {
  const pool = new pg.Pool({ connectionString: url });
  const db = drizzle(pool, { schema });
  return { db, pool };
}

export async function migrateDb(url: string) {
  const { db, pool } = createDb(url);
  try {
    await migrate(db, { migrationsFolder });
  } finally {
    await pool.end();
  }
}

export type Db = ReturnType<typeof createDb>["db"];
```

```ts
export function hashEmbed(text: string, dim = 256): number[] {
  const v = new Array<number>(dim).fill(0);
  for (const tok of text.toLowerCase().match(/[a-z0-9]+/g) ?? []) {
    let h = 2166136261;
    for (let i = 0; i < tok.length; i++) {
      h ^= tok.charCodeAt(i);

    }
    v[Math.abs(h) % dim] += 1;
  }
```

```ts
import PgBoss from "pg-boss";
import { createDb, type Db } from "@ragbench/db";

export type JobHandler<T> = (data: T, ctx: { db: Db }) => Promise<void>;

export async function startWorker(opts: {
  databaseUrl: string;
  handlers: Record<string, JobHandler<any>>;
}) {
  const boss = new PgBoss(opts.databaseUrl);
  const { db, pool } = createDb(opts.databaseUrl);
  await boss.start();

  for (const [name, handler] of Object.entries(opts.handlers)) {
    await boss.createQueue(name, { retryLimit: 3, retryBackoff: true } as PgBoss.Queue);
    await boss.work(name, async ([job]) => {
      await handler(job.data, { db });
    });
  }

  return {
    boss,
    async stop() {
      await boss.stop({ graceful: true, wait: true });
      await pool.end();
    },
  };
}

export async function enqueue(
  boss: PgBoss,
  name: string,
  data: object,
  singletonKey?: string,
): Promise<string | null> {
  return boss.send(name, data, singletonKey ? { singletonKey } : {});
}
```

```ts
import bcrypt from "bcryptjs";
import { eq } from "drizzle-orm";
import { organizations, users } from "@ragbench/db";
import { getDb } from "@/lib/db";

  email: z.string().email(),
  password: z.string().min(8),
  organizationName: z.string().min(1),

export async function POST(req: Request) {
  const parsed = Body.safeParse(await req.json().catch(() => null));

  const { email, password, organizationName } = parsed.data;

    const [org] = await tx.insert(organizations).values({ name: organizationName }).returning();
    const [user] = await tx.insert(users).values({
      organizationId: org.id, email: email.toLowerCase(), passwordHash,
    }).returning();
    return { userId: user.id, organizationId: org.id };
  });
```

```ts
import NextAuth from "next-auth";
import Credentials from "next-auth/providers/credentials";
import { verifyCredentials } from "@/auth-core";

export const { handlers, auth, signIn, signOut } = NextAuth({
  session: { strategy: "jwt" },
  providers: [
    Credentials({
      credentials: { email: {}, password: {} },
      async authorize(credentials) {
        const email = String(credentials?.email ?? "");
        const password = String(credentials?.password ?? "");
        const user = await verifyCredentials(email, password);
        if (!user) return null;
        return { id: user.id, email, organizationId: user.organizationId } as any;
      },
    }),
  ],
  callbacks: {
    jwt({ token, user }) {
      if (user) {
        token.userId = (user as any).id;
        token.organizationId = (user as any).organizationId;
      }
      return token;
    },
    session({ session, token }) {
      (session.user as any).id = token.userId;
      (session.user as any).organizationId = token.organizationId;
      return session;
    },
  },
});
```

`apps/web/src/app/api/projects/route.ts`:
```ts
import { NextResponse } from "next/server";
import { z } from "zod";
import { eq } from "drizzle-orm";
import { projects } from "@ragbench/db";
import { getDb } from "@/lib/db";
import { auth } from "@/auth";

type Session = { user: { id: string; organizationId: string } } | null;

export async function listProjects(session: Session) {
  if (!session?.user?.organizationId) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }
  const rows = await getDb().select().from(projects)
    .where(eq(projects.organizationId, session.user.organizationId));
  return NextResponse.json({ projects: rows });
}

export async function createProject(req: Request, session: Session) {
  if (!session?.user?.organizationId) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }
  const parsed = z.object({ name: z.string().min(1) }).safeParse(await req.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "invalid payload" }, { status: 400 });
  const [project] = await getDb().insert(projects).values({
    organizationId: session.user.organizationId, name: parsed.data.name,
  }).returning();
  return NextResponse.json({ project }, { status: 201 });
}

export async function GET() {
  return listProjects((await auth()) as Session);
}

export async function POST(req: Request) {
  return createProject(req, (await auth()) as Session);
}
```
